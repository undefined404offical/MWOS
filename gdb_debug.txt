# GDB 调试脚本 - 调试 syscall #GP 错误
target remote :1234

# 设置断点
break syscall_entry
break idt_handler
break handle_syscall

# 继续执行
continue

# 当命中断点时打印信息
commands 1
  echo \n=== HIT syscall_entry ===\n
  info registers
  bt
  continue
end

commands 2
  echo \n=== HIT idt_handler (EXCEPTION) ===\n
  info registers
  bt
  continue
end

commands 3
  echo \n=== HIT handle_syscall ===\n
  info registers
  bt
  continue
end
